Table Definitions
studentsPRIMARY
CREATE TABLE students (
  student_id    INT PRIMARY KEY AUTO_INCREMENT,
  roll_no       VARCHAR(20) UNIQUE NOT NULL,
  full_name     VARCHAR(100) NOT NULL,
  department    VARCHAR(60) NOT NULL,
  semester      TINYINT NOT NULL,
  section       CHAR(1),
  email         VARCHAR(100) UNIQUE,
  dob           DATE,
  enrolled_on   DATE DEFAULT CURRENT_DATE
);
subjectsLOOKUP
CREATE TABLE subjects (
  subject_id    INT PRIMARY KEY AUTO_INCREMENT,
  subject_code  VARCHAR(20) UNIQUE NOT NULL,
  subject_name  VARCHAR(100) NOT NULL,
  department    VARCHAR(60),
  semester      TINYINT,
  credits       TINYINT,
  max_internal  INT DEFAULT 30,
  max_external  INT DEFAULT 70
);
attendanceTRACKING
CREATE TABLE attendance (
  att_id        INT PRIMARY KEY AUTO_INCREMENT,
  student_id    INT,
  subject_id    INT,
  total_classes INT NOT NULL,
  attended      INT NOT NULL,
  att_percent   DECIMAL(5,2) AS (
    ROUND(attended/NULLIF(total_classes,0)*100,2)
  ) STORED,
  FOREIGN KEY(student_id) REFERENCES students(student_id),
  FOREIGN KEY(subject_id) REFERENCES subjects(subject_id)
);
marksCORE
CREATE TABLE marks (
  mark_id       INT PRIMARY KEY AUTO_INCREMENT,
  student_id    INT,
  subject_id    INT,
  internal_marks DECIMAL(5,2) DEFAULT 0,
  external_marks DECIMAL(5,2) DEFAULT 0,
  total_marks   DECIMAL(6,2) AS (
    internal_marks + external_marks
  ) STORED,
  percentage    DECIMAL(5,2),  -- updated by trigger
  grade         VARCHAR(3),    -- updated by trigger
  exam_year     YEAR,
  FOREIGN KEY(student_id) REFERENCES students(student_id),
  FOREIGN KEY(subject_id) REFERENCES subjects(subject_id)
);
Grade Scale & Trigger
Auto Grade TriggerTRIGGER
DELIMITER $$
CREATE TRIGGER calc_grade_percent
BEFORE INSERT ON marks
FOR EACH ROW
BEGIN
  DECLARE max_tot INT;
  SELECT (max_internal + max_external) INTO max_tot
  FROM subjects WHERE subject_id = NEW.subject_id;

  SET NEW.percentage = ROUND(
    (NEW.internal_marks + NEW.external_marks) / max_tot * 100, 2);

  SET NEW.grade = CASE
    WHEN NEW.percentage >= 90 THEN 'O'
    WHEN NEW.percentage >= 80 THEN 'A+'
    WHEN NEW.percentage >= 70 THEN 'A'
    WHEN NEW.percentage >= 60 THEN 'B+'
    WHEN NEW.percentage >= 50 THEN 'B'
    WHEN NEW.percentage >= 40 THEN 'C'
    ELSE 'F'
  END;
END$$
DELIMITER ;

Q03 Full marksheet for a student SELECT ▶
Get subject-wise internal, external, total, percentage, and grade for one student.
SELECT
  sub.subject_code,
  sub.subject_name,
  m.internal_marks,
  sub.max_internal                      AS max_int,
  m.external_marks,
  sub.max_external                      AS max_ext,
  m.total_marks,
  (sub.max_internal + sub.max_external) AS max_total,
  m.percentage,
  m.grade
FROM marks m
JOIN subjects sub ON m.subject_id = sub.subject_id
WHERE m.student_id = 1
ORDER BY sub.subject_code;
Q04 Overall percentage & CGPA for all students SELECT ▶
Aggregate total marks, overall percentage, rank, and pass/fail status.
SELECT
  s.roll_no,
  s.full_name,
  s.department,
  s.semester,
  SUM(m.total_marks)                    AS total_obtained,
  SUM(sub.max_internal + sub.max_external) AS grand_total,
  ROUND(SUM(m.total_marks) /
    SUM(sub.max_internal + sub.max_external) * 100, 2) AS overall_pct,
  RANK() OVER
    (PARTITION BY s.semester ORDER BY
      SUM(m.total_marks) DESC)            AS sem_rank,
  CASE WHEN MIN(m.percentage) >= 40
    THEN 'PASS' ELSE 'FAIL' END          AS result
FROM students s
JOIN marks m    ON s.student_id = m.student_id
JOIN subjects sub ON m.subject_id = sub.subject_id
GROUP BY s.student_id
ORDER BY overall_pct DESC;
Q05 Attendance shortfall (below 75%) SELECT ▶
Find students with low attendance in any subject, and how many classes they need to reach 75%.
SELECT
  s.roll_no,
  s.full_name,
  sub.subject_name,
  a.attended,
  a.total_classes,
  a.att_percent,
  GREATEST(0,
    CEIL((0.75 * a.total_classes - a.attended)
      / (1 - 0.75))
  ) AS classes_needed_for_75
FROM attendance a
JOIN students s  ON a.student_id = s.student_id
JOIN subjects sub ON a.subject_id = sub.subject_id
WHERE a.att_percent < 75
ORDER BY a.att_percent ASC;
Q06 Subject-wise topper & average SELECT ▶
Find the highest scorer and average marks per subject.
SELECT
  sub.subject_code,
  sub.subject_name,
  MAX(m.total_marks)        AS highest,
  MIN(m.total_marks)        AS lowest,
  ROUND(AVG(m.total_marks),2) AS avg_marks,
  ROUND(AVG(m.percentage),2)  AS avg_pct,
  COUNT(m.mark_id)          AS students_appeared
FROM marks m
JOIN subjects sub ON m.subject_id = sub.subject_id
GROUP BY sub.subject_id
ORDER BY avg_pct DESC;
Q07 Grade distribution count SELECT ▶
Count how many students got each grade across all subjects.
SELECT
  grade,
  COUNT(*) AS total_count,
  ROUND(COUNT(*) * 100.0 /
    (SELECT COUNT(*) FROM marks), 1) AS percentage_share
FROM marks
GROUP BY grade
ORDER BY FIELD(grade, 'O','A+','A','B+','B','C','F');

Views & Stored Procedures
V01 VIEW — Full Student Report Card VIEW ▶
A reusable view joining all tables for report card generation.
CREATE VIEW vw_student_report AS
SELECT
  s.roll_no,
  s.full_name,
  s.department,
  s.semester,
  sub.subject_code,
  sub.subject_name,
  sub.credits,
  m.internal_marks,
  m.external_marks,
  m.total_marks,
  m.percentage,
  m.grade,
  a.att_percent,
  CASE
    WHEN m.percentage >= 40 AND a.att_percent >= 75
      THEN 'ELIGIBLE'
    WHEN a.att_percent < 75
      THEN 'DETAINED'
    ELSE 'FAIL'
  END AS exam_status
FROM students s
JOIN marks      m   ON s.student_id   = m.student_id
JOIN subjects   sub ON m.subject_id   = sub.subject_id
LEFT JOIN attendance a
  ON s.student_id = a.student_id
 AND m.subject_id = a.subject_id;
V02 VIEW — Department Toppers VIEW ▶
Shows rank-1 student per department per semester using window functions.
CREATE VIEW vw_dept_toppers AS
WITH ranked AS (
  SELECT
    s.department, s.semester,
    s.roll_no, s.full_name,
    ROUND(SUM(m.total_marks) /
      SUM(sub.max_internal + sub.max_external)*100, 2) AS pct,
    RANK() OVER (
      PARTITION BY s.department, s.semester
      ORDER BY SUM(m.total_marks) DESC
    ) AS rnk
  FROM students s
  JOIN marks m    ON s.student_id = m.student_id
  JOIN subjects sub ON m.subject_id = sub.subject_id
  GROUP BY s.student_id
)
SELECT * FROM ranked WHERE rnk = 1;
P01 PROCEDURE — Get Student Report Card PROCEDURE ▶
Pass a roll number to get a formatted report card for that student.
DELIMITER $$
CREATE PROCEDURE sp_GetReportCard(
  IN  p_roll_no  VARCHAR(20),
  IN  p_year     YEAR
)
BEGIN
  SELECT
    r.subject_code,
    r.subject_name,
    r.credits,
    r.internal_marks,
    r.external_marks,
    r.total_marks,
    r.percentage,
    r.grade,
    r.att_percent,
    r.exam_status
  FROM vw_student_report r
  JOIN students s ON s.roll_no = r.roll_no
  JOIN marks    m ON m.student_id = s.student_id
  WHERE r.roll_no = p_roll_no
    AND m.exam_year = p_year;
END$$
DELIMITER ;
-- Usage:
CALL sp_GetReportCard('2024ECE001', 2024);

analysis_queries.sql 
-- Year-over-Year Rainfall Change by Region
SELECT
  region,
  year,
  avg_rainfall_mm,
  LAG(avg_rainfall_mm) OVER (
    PARTITION BY region
    ORDER BY year
  ) AS prev_year,
  ROUND(
    avg_rainfall_mm -
    LAG(avg_rainfall_mm) OVER (
      PARTITION BY region ORDER BY year
    ), 2
  ) AS yoy_change
FROM climate_data
ORDER BY region, year;

-- Top 3 Anomaly Years per Region
WITH ranked AS (
  SELECT
    region, year, avg_temperature_c,
    RANK() OVER (
      PARTITION BY region
      ORDER BY ABS(
        avg_temperature_c -
        AVG(avg_temperature_c) OVER
          (PARTITION BY region)
      ) DESC
    ) AS anomaly_rank
  FROM climate_data
)
SELECT * FROM ranked
WHERE anomaly_rank <= 3;

queries.sql 
-- Top Performer per Department
WITH ranked_students AS (
  SELECT
    s.student_id, s.name, d.dept_name,
    AVG(CASE e.grade
      WHEN 'A' THEN 10
      WHEN 'B' THEN 8
      WHEN 'C' THEN 6
      ELSE 0
    END) AS gpa,
    RANK() OVER (
      PARTITION BY d.dept_name
      ORDER BY AVG(CASE e.grade
        WHEN 'A' THEN 10
        WHEN 'B' THEN 8
        ELSE 0
      END) DESC
    ) AS dept_rank
  FROM students s
  JOIN departments d ON s.dept_id = d.dept_id
  JOIN enrollments e ON s.student_id = e.student_id
  GROUP BY s.student_id, d.dept_name
)
SELECT * FROM ranked_students
WHERE dept_rank = 1;

analyze.py 
import pandas as pd
import matplotlib.pyplot as plt

# Load and merge multi-source data
gdp = pd.read_csv('data/raw/gdp_growth.csv')
cpi = pd.read_csv('data/raw/cpi_inflation.csv')
merged = gdp.merge(cpi, on='year')

# Null handling
merged.fillna(merged.mean(numeric_only=True),
              inplace=True)

# Rolling average + YoY change
merged['gdp_rolling'] = (
  merged['gdp_growth']
  .rolling(window=3)
  .mean()
)
merged['gdp_yoy'] = (
  merged['gdp_growth']
  .pct_change() * 100
)

# Decade-level volatility
merged['decade'] = (
  merged['year'] // 10
) * 10
vol = (merged
  .groupby('decade')['gdp_growth']
  .std()
  .reset_index()
)

# Annotated chart
fig, ax = plt.subplots(figsize=(12, 5))
ax.plot(merged['year'],
        merged['gdp_growth'],
        color='#1a5276', linewidth=2)
ax.axvline(2008, color='red',
           alpha=0.4, linestyle=':')
ax.text(2008.2, 9, 'GFC',
       fontsize=8, color='red')

obstacle_detection.cpp 
#include <opencv2/opencv.hpp>
#include <iostream>

int main() {
  cv::VideoCapture cap(0);
  cv::Mat frame, gray,
           blurred, edges;

  while(true) {
    cap >> frame;

    // Preprocessing
    cv::cvtColor(frame, gray,
      cv::COLOR_BGR2GRAY);
    cv::GaussianBlur(gray, blurred,
      cv::Size(5,5), 0);
    cv::Canny(blurred, edges,
      50, 150);

    // Contour detection
    std::vector<std::vector<
      cv::Point>> contours;
    cv::findContours(edges,
      contours,
      cv::RETR_EXTERNAL,
      cv::CHAIN_APPROX_SIMPLE);

    for (auto& c : contours) {
      double area =
        cv::contourArea(c);
      if (area > THRESHOLD) {
        triggerAlert();
      }
    }
  }
  return 0;
}
